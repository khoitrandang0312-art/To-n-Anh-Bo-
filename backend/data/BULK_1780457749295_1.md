answers:
  - "$x=2; x=3$"

  - "$x=2; x=-3$"
  - "$x=-2; x=-3$"
solution: |
  Ta có phương trình tương đương $(x-2)(x-3) = 0$.
  Suy ra $x_1 = 2, x_2 = 3$.
